\section{Applicazioni Lineari}

\begin{concept}{Definizione}
Una funzione $f: V \to W$ è lineare se:
\begin{itemize}
    \item $f(v_1 + v_2) = f(v_1) + f(v_2)$
    \item $f(\lambda v) = \lambda f(v)$
\end{itemize}
\end{concept}

\subsection{Nucleo e Immagine}
\begin{concept}{Teorema della Dimensione}
$\dim(V) = \dim(\ker(f)) + \dim(\text{Im}(f))$
\end{concept}
